1. FLITERING SQL QUERY 
SELECT * 
FROM GymMembership 
WHERE Membership_Type = 'Basic' AND Member_Gender = 'Female' AND Health_risk_flag = 'High';
------------------------------------------------------------------
2.AGGREGATION SQL QUERY 
SELECT SUM(member_age) AS "Sum of Member Ages"
FROM GymMembership
----------------------------------------------------------------
3. SORTING QUERY 
SELECT * FROM GymMembership
ORDER BY Member_Age DESC;
-------------------------------------------------------------
4. LIMIT QUERY 
SELECT * 
FROM GymMembership 
LIMIT 3;
-------------------------------------------------------------
5. MAX Query 
SELECT MAX(monthly_fee) 
FROM GymMembership
-----------------------------------------------------------
